rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isOwner() { return signedIn() && request.auth.uid == "wC4l6PFQA8csi1DZTwsQEMuwOAk1"; }
    function profilePath(uid) {
      return /databases/$(database)/documents/userProfiles/$(uid);
    }
    function isApproved(uid) {
      return signedIn()
        && exists(profilePath(uid))
        && get(profilePath(uid)).data.status == "approved";
    }
    function hasAnyWritePermission(uid) {
      let p = get(profilePath(uid)).data.permissions;
      return p.get("all", false) == true
        || p.get("Sales", {}).get("add", false) == true
        || p.get("Sales", {}).get("edit", false) == true
        || p.get("Sales", {}).get("delete", false) == true
        || p.get("Invoices", {}).get("add", false) == true
        || p.get("Invoices", {}).get("edit", false) == true
        || p.get("Invoices", {}).get("delete", false) == true
        || p.get("Stock", {}).get("add", false) == true
        || p.get("Stock", {}).get("edit", false) == true
        || p.get("Stock", {}).get("delete", false) == true
        || p.get("Market Credit", {}).get("add", false) == true
        || p.get("Market Credit", {}).get("edit", false) == true
        || p.get("Market Credit", {}).get("delete", false) == true
        || p.get("Recovery", {}).get("add", false) == true
        || p.get("Recovery", {}).get("edit", false) == true
        || p.get("Recovery", {}).get("delete", false) == true
        || p.get("Expenses", {}).get("add", false) == true
        || p.get("Expenses", {}).get("edit", false) == true
        || p.get("Expenses", {}).get("delete", false) == true
        || p.get("User Management", {}).get("approve", false) == true
        || p.get("Settings", {}).get("edit", false) == true
        || p.get("Settings", {}).get("delete", false) == true;
    }

    // Users can create only their own pending registration. Only Owner can approve/change profiles.
    match /userProfiles/{userId} {
      allow read: if isOwner() || (signedIn() && request.auth.uid == userId);
      allow create: if signedIn() && request.auth.uid == userId
        && request.resource.data.status == "pending"
        && request.resource.data.role == "pending"
        && request.resource.data.permissions is map
        && request.resource.data.keys().hasOnly([
          "uid", "email", "requestedName", "displayName",
          "status", "role", "permissions", "createdAt"
        ]);
      allow create, update, delete: if isOwner();
    }

    // Shared AL QADIR TRADERS company data. Owner controls initialization;
    // approved users may read, and users with at least one write permission may write.
    match /companies/al-qadir-traders/agency/main {
      allow read: if isOwner() || isApproved(request.auth.uid);
      allow create, update: if isOwner() || (
        isApproved(request.auth.uid)
        && hasAnyWritePermission(request.auth.uid)
        && request.resource.data.ownerUid == "wC4l6PFQA8csi1DZTwsQEMuwOAk1"
        && request.resource.data.companyId == "al-qadir-traders"
      );
      allow delete: if isOwner();
    }

    // Retain the existing per-user backup path for manual recovery.
    match /users/{userId}/agency/{backupId} {
      allow read: if isOwner() || (signedIn()
        && request.auth.uid == userId
        && backupId == "main"
        && isApproved(userId));
      allow create, update: if isOwner() || (signedIn()
        && request.auth.uid == userId
        && backupId == "main"
        && isApproved(userId)
        && request.resource.data.ownerUid == userId);
      allow delete: if isOwner();
    }

    match /{document=**} { allow read, write: if false; }
  }
}
